    function handleSubmit(e){

      e.preventDefault();

      const data = Object.fromEntries(new FormData(e.target).entries());


      // Simple "mailto" handoff (no backend).

      const subject = encodeURIComponent(`Inquiry — ${data.name}${data.company ? " / " + data.company : ""}`);

      const body = encodeURIComponent(

        `Name: ${data.name}\nEmail: ${data.email}\nCompany/Project: ${data.company || "-"}\nLocation: ${data.location || "-"}\n\nMessage:\n${data.message}\n`

      );


      // TODO: Replace with your target email address:

      const to = "info@yourdomain.com";


      window.location.href = `mailto:${to}?subject=${subject}&body=${body}`;

      return false;

    }
